const fs = require('node:fs');
const path = require('node:path');
const Module = require('node:module');
const script = '/Users/thinkstu/Desktop/copy/FluentRead-harness-actions-audio-20260905/scripts/run-harness-reading-test.cjs';
if (!process.argv.includes('--persistence-only')) throw new Error('集成诊断只允许 --persistence-only');
const source = fs.readFileSync(script, 'utf8');
const marker = '        result.backgroundSaveReceipts = await worker.evaluate(() => globalThis.__fluentReadHarnessSaveReceipts);';
if (source.split(marker).length !== 2) throw new Error('无法唯一定位持久化验证结束边界');
const smoke = `
        result.diagnosticSubset = 'unchanged persistence-only 3 cases plus HTML modal reading/audio smoke';
        await page.reload({waitUntil: 'domcontentloaded'});
        await page.waitForTimeout(700);
        await page.evaluate(() => {
            const dialog = document.createElement('dialog');
            dialog.id = 'harness-integration-modal';
            dialog.style.cssText = 'width:620px;max-width:calc(100vw - 80px);padding:32px;border:1px solid #aaa;border-radius:16px;';
            const heading = document.createElement('h2');
            heading.textContent = 'Reading inside a modal';
            const sentence = document.createElement('p');
            sentence.id = 'harness-modal-target';
            sentence.style.cssText = 'display:inline-block;white-space:nowrap;margin:20px 0 320px;';
            sentence.textContent = 'Although the task was difficult, she finished it on time.';
            dialog.append(heading, sentence);
            document.body.append(dialog);
            dialog.showModal();
        });
        const modalSentence = page.locator('#harness-modal-target');
        const modalBox = await modalSentence.boundingBox();
        assert(modalBox, 'HTML modal 原文无可选几何位置');
        await page.mouse.move(modalBox.x + 1, modalBox.y + modalBox.height / 2);
        await page.mouse.down();
        await page.mouse.move(modalBox.x + modalBox.width - 1, modalBox.y + modalBox.height / 2, {steps: 12});
        await page.mouse.up();
        await page.waitForTimeout(450);
        const modalSelection = await page.evaluate(() => getSelection()?.toString().trim());
        assert(modalSelection === await modalSentence.innerText(), 'HTML modal 没有选中完整原句');
        const beforeModalRequest = requests.length;
        await clickShadowButton(page, '读懂');
        await waitForReadingComplete(page);
        assert(requests.length === beforeModalRequest + 1 && (await currentReadingAnswer(page)).includes('大意'), 'HTML modal 阅读没有走真实 provider 或没有完成回答');
        assert(await page.evaluate(() => {
            const dialog = document.querySelector('#harness-integration-modal');
            return dialog.matches(':modal') && dialog.contains(document.querySelector('#fluent-read-selection-translator-container'));
        }), '阅读卡没有在真实 modal 内接管挂载位置');
        const beforeModalSpeech = ttsRequests.length;
        await clickShadowButton(page, '朗读原文');
        await waitForShadowButton(page, '停止朗读');
        await waitUntil(() => ttsRequests.length === beforeModalSpeech + 1, 'modal 原文朗读没有走真实 TTS 合成');
        assert(ttsRequests.at(-1).ssml.includes(modalSelection), 'modal TTS 原文不一致');
        await page.waitForTimeout(250);
        await clickShadowButton(page, '停止朗读');
        await waitForShadowButton(page, '朗读原文');
        const modalScreenshot = path.join(args.artifactsDir, 'harness-html-modal-reading-audio.png');
        await page.screenshot({path: modalScreenshot}); result.screenshots.push(modalScreenshot);
        await page.evaluate(() => document.querySelector('#harness-integration-modal').close());
        record('html-modal-reading-and-source-audio-smoke', 'passed', {selectedText: modalSelection, providerCalls: requests.length - beforeModalRequest, synthesisRequests: ttsRequests.length - beforeModalSpeech, nativeHtmlDialog: true, muted: true, listeningVerified: false});
`;
const runner = new Module(script, module);
runner.filename = script;
runner.paths = Module._nodeModulePaths(path.dirname(script));
runner._compile(source.replace(marker, smoke + '\n' + marker), script);
